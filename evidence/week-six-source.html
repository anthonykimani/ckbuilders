resource Escrow has store, create, consume, burn { escrow_id: Hash, payer: Address, recipient: Address, escrow_lock: Address, payment_hash: Hash, refund_height: u64, locked_capacity: u64, state: EscrowState, } receipt SettlementRecord has create { escrow_id: Hash, action: u8, settled_by: Address, settled_at: u64, } flow Escrow.state { initial Pending; terminal Claimed, Refunded; Pending -> Claimed; Pending -> Refunded; } action claim( escrow: Escrow, preimage: Hash, claimed_by: Address, ) -> (settled: Escrow, record: SettlementRecord) { transition escrow.state: Pending -> settled.state: Claimed verification let now = env::current_timepoint() require escrow.state == EscrowState::Pending, "escrow is not pending" require claimed_by == escrow.recipient, "only recipient can claim" require hash_blake2b(preimage) == escrow.payment_hash, "wrong preimage" require now < escrow.refund_height, "claim window expired" consume escrow create settled = Escrow { ..., state: EscrowState::Claimed } with_lock(escrow.recipient) std::cell::preserve_capacity(settled, escrow) }
